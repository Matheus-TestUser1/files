' === SISTEMA PDV MADEIREIRA COM UNIDADES PADRONIZADAS ===
Private Const QTD_MINIMA As Long = 1
Private Const QTD_MAXIMA As Long = 9999

' === VARIÁVEIS GLOBAIS PARA CONTROLE DE TOTAIS ===
Private totalSubtotal As Currency
Private totalDescontos As Currency
Private totalFinal As Currency
Private quantidadeItens As Long

' === FUNÇÃO PARA FORMATAR QUANTIDADE NO PADRÃO PDV MADEIREIRA ===
Private Function FormatarQuantidadePDV(quantidade As Long) As String
    FormatarQuantidadePDV = Format(quantidade, "0") & ",000"
End Function

' === FUNÇÃO PARA CONVERTER QUANTIDADE PDV PARA REAL ===
Private Function QuantidadePDVParaReal(quantidadePDV As String) As Long
    On Error Resume Next
    Dim temp As String
    temp = Trim(CStr(quantidadePDV))
    
    If temp = "" Then
        QuantidadePDVParaReal = 1
        Exit Function
    End If
    
    ' CORREÇÃO: Se terminar com ",000" é formato visual PDV
    If Right(temp, 4) = ",000" Then
        temp = Left(temp, Len(temp) - 4)
    End If
    
    temp = Trim(temp)
    
    If IsNumeric(temp) And temp <> "" Then
        QuantidadePDVParaReal = CLng(temp)
    Else
        QuantidadePDVParaReal = 1
    End If
    On Error GoTo 0
End Function

' === EVENTO DE DUPLO CLIQUE PARA EDITAR QUANTIDADE ===
Private Sub lstSelecionados_DblClick(ByVal Cancel As MSForms.ReturnBoolean)
    On Error Resume Next
    If Me.lstSelecionados.ListIndex < 0 Then Exit Sub
    Call EditarQuantidadeItem(Me.lstSelecionados.ListIndex)
    On Error GoTo 0
End Sub

' === FUNÇÃO PARA EDITAR QUANTIDADE DE ITEM ===
Private Sub EditarQuantidadeItem(indiceItem As Long)
    On Error GoTo TratarErro
    
    Dim nomeItem As String
    Dim qtdDisplayAtual As String
    Dim qtdRealAtual As Long
    Dim precoUnitario As Currency
    Dim descontoAtual As Currency
    
    nomeItem = CStr(Me.lstSelecionados.List(indiceItem, 1))
    qtdDisplayAtual = CStr(Me.lstSelecionados.List(indiceItem, 4))
    qtdRealAtual = QuantidadePDVParaReal(CStr(Me.lstSelecionados.List(indiceItem, 4)))
    precoUnitario = TextoParaMoeda(CStr(Me.lstSelecionados.List(indiceItem, 3)))
    descontoAtual = TextoParaMoeda(CStr(Me.lstSelecionados.List(indiceItem, 5)))
    
    Dim novaQtdStr As String
    Dim qtdRealNova As Long
    
    novaQtdStr = InputBox( _
        "ALTERAR QUANTIDADE - SISTEMA PDV MADEIREIRA" & vbCrLf & vbCrLf & _
        "Produto: " & nomeItem & vbCrLf & _
        "Preço: " & Format(precoUnitario, "R$ #,##0.00") & " cada" & vbCrLf & _
        "Quantidade Atual: " & qtdRealAtual & " produto(s)" & vbCrLf & _
        "Display PDV: " & qtdDisplayAtual & " UN" & vbCrLf & vbCrLf & _
        "Digite quantos produtos:", _
        "Quantidade Real de Produtos", _
        qtdRealAtual)
    
    If novaQtdStr = "" Then Exit Sub
    
    If Not IsNumeric(novaQtdStr) Then
        MsgBox "Digite apenas números inteiros!", vbExclamation
        Exit Sub
    End If
    
    qtdRealNova = CLng(novaQtdStr)
    
    If qtdRealNova < QTD_MINIMA Or qtdRealNova > QTD_MAXIMA Then
        MsgBox "Quantidade deve estar entre " & QTD_MINIMA & " e " & QTD_MAXIMA, vbExclamation
        Exit Sub
    End If
    
    If qtdRealNova = qtdRealAtual Then
        MsgBox "Quantidade não foi alterada.", vbInformation
        Exit Sub
    End If
    
    Dim qtdDisplayNova As String
    Dim novoSubtotal As Currency
    Dim novoTotal As Currency
    
    qtdDisplayNova = FormatarQuantidadePDV(qtdRealNova)
    novoSubtotal = precoUnitario * qtdRealNova
    novoTotal = novoSubtotal - descontoAtual
    
    If descontoAtual > novoSubtotal Then
        descontoAtual = novoSubtotal
        novoTotal = 0
    End If
    
    Dim confirmacao As VbMsgBoxResult
    confirmacao = MsgBox( _
        "CONFIRMAR ALTERAÇÃO PDV" & vbCrLf & vbCrLf & _
        nomeItem & vbCrLf & vbCrLf & _
        "Produtos: " & qtdRealAtual & " -> " & qtdRealNova & vbCrLf & _
        "Display: " & qtdDisplayAtual & " -> " & qtdDisplayNova & vbCrLf & _
        "Subtotal: " & Format(precoUnitario * qtdRealAtual, "R$ #,##0.00") & " -> " & Format(novoSubtotal, "R$ #,##0.00") & vbCrLf & _
        "Total: " & Format(novoTotal, "R$ #,##0.00") & vbCrLf & vbCrLf & _
        "Confirma?", vbYesNo + vbQuestion, "PDV Madeireira")
    
    If confirmacao = vbNo Then Exit Sub
    
    Me.lstSelecionados.List(indiceItem, 4) = qtdDisplayNova
    Me.lstSelecionados.List(indiceItem, 5) = Format(descontoAtual, "R$ #,##0.00")
    Me.lstSelecionados.List(indiceItem, 6) = Format(novoTotal, "R$ #,##0.00")
    
    Call CalcularTodosOsTotais
    
    MsgBox "QUANTIDADE ALTERADA!" & vbCrLf & vbCrLf & _
           nomeItem & vbCrLf & _
           "Produtos: " & qtdRealNova & vbCrLf & _
           "Display PDV: " & qtdDisplayNova & " UN" & vbCrLf & _
           "Total: " & Format(novoTotal, "R$ #,##0.00"), vbInformation
    
    Exit Sub
    
TratarErro:
    MsgBox "Erro ao alterar quantidade: " & Err.Description, vbCritical
End Sub

' === BOTÃO ADICIONAR MODIFICADO PARA PDV MADEIREIRA ===
Private Sub btnAdicionar_Click()
    On Error GoTo TratarErro
    
    If Me.lstProdutos.ListIndex < 0 Then
        MsgBox "Selecione um produto.", vbExclamation
        Exit Sub
    End If
    
    Dim i As Long, idProduto As String
    idProduto = Me.lstProdutos.List(Me.lstProdutos.ListIndex, 0)
    
    For i = 0 To Me.lstSelecionados.ListCount - 1
        If CStr(Me.lstSelecionados.List(i, 0)) = idProduto Then
            MsgBox "Produto já adicionado.", vbInformation
            Exit Sub
        End If
    Next i
    
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Worksheets("Produtos")
    
    Dim precoUnitario As Currency
    Dim linhaAtual As Long
    Dim produtoEncontrado As Boolean
    
    produtoEncontrado = False
    For linhaAtual = 2 To ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
        If CStr(ws.Cells(linhaAtual, 1).Value) = idProduto Then
            precoUnitario = ws.Cells(linhaAtual, 6).Value
            produtoEncontrado = True
            Exit For
        End If
    Next linhaAtual
    
    If Not produtoEncontrado Then
        MsgBox "Produto não encontrado na base de dados!", vbExclamation
        Exit Sub
    End If
    
    Dim qtdReal As Long
    qtdReal = CLng(Me.txtQuantidade.Value)
    
    Dim qtdDisplay As String
    qtdDisplay = FormatarQuantidadePDV(qtdReal)
    
    Dim subtotalItem As Currency
    subtotalItem = precoUnitario * qtdReal
    
    With Me.lstSelecionados
        .AddItem
        .List(.ListCount - 1, 0) = idProduto
        .List(.ListCount - 1, 1) = Me.lstProdutos.List(Me.lstProdutos.ListIndex, 1)
        .List(.ListCount - 1, 2) = "UN"
        .List(.ListCount - 1, 3) = Format(precoUnitario, "R$ #,##0.00")
        .List(.ListCount - 1, 4) = qtdDisplay
        .List(.ListCount - 1, 5) = "R$ 0,00"
        .List(.ListCount - 1, 6) = Format(subtotalItem, "R$ #,##0.00")
    End With
    
    Call CalcularTodosOsTotais
    
    Me.txtQuantidade.Value = QTD_MINIMA
    
    MsgBox "Produto adicionado!" & vbCrLf & vbCrLf & _
           "Quantidade: " & qtdDisplay & " UN" & vbCrLf & _
           "Preço: " & Format(precoUnitario, "R$ #,##0.00") & " cada" & vbCrLf & _
           "Total: " & Format(subtotalItem, "R$ #,##0.00") & vbCrLf & vbCrLf & _
           "(" & qtdReal & " produto(s) real)", vbInformation
    
    Exit Sub
    
TratarErro:
    MsgBox "Erro ao adicionar produto: " & Err.Description, vbCritical
End Sub

' === BOTÃO REMOVER ===
Private Sub btnRemover_Click()
    On Error Resume Next
    
    If Me.lstSelecionados.ListIndex >= 0 Then
        Dim itemRemovido As String
        itemRemovido = Me.lstSelecionados.List(Me.lstSelecionados.ListIndex, 1)
        
        Me.lstSelecionados.RemoveItem Me.lstSelecionados.ListIndex
        Call CalcularTodosOsTotais

    End If
    
    On Error GoTo 0

' === SISTEMA DE DESCONTO INTEGRADO ===
Private Sub btnAplicarDesconto_Click()
    On Error GoTo TratarErro
    
    If Me.lstSelecionados.ListIndex < 0 Then
        MsgBox "Selecione um produto para aplicar desconto.", vbExclamation
        Exit Sub
    End If
    
    Dim valorDesconto As Currency
    
    On Error Resume Next
    valorDesconto = TextoParaMoeda(Me.lbldesc1.Value)
    
    If valorDesconto < 0 Then
        MsgBox "Valor de desconto não pode ser negativo.", vbExclamation
        Exit Sub
    End If
    On Error GoTo 0
    
    Call AplicarDescontoItem(Me.lstSelecionados.ListIndex, valorDesconto)
    
    Me.lbldesc1.Value = ""
    Call CalcularTodosOsTotais
    
    Exit Sub
    
TratarErro:
    MsgBox "Erro ao aplicar desconto: " & Err.Description, vbCritical
End Sub

' === FUNÇÃO PARA APLICAR DESCONTO EM ITEM ===
Private Sub AplicarDescontoItem(indiceItem As Long, valorDesconto As Currency)
    On Error Resume Next
    
    Dim precoUnitario As Currency
    Dim qtdReal As Long
    Dim descontoAplicado As Currency
    Dim subtotalItem As Currency
    Dim novoTotal As Currency
    
    precoUnitario = TextoParaMoeda(CStr(Me.lstSelecionados.List(indiceItem, 3)))
    qtdReal = QuantidadePDVParaReal(CStr(Me.lstSelecionados.List(indiceItem, 4)))
    subtotalItem = precoUnitario * qtdReal
    
    descontoAplicado = valorDesconto
    
    If descontoAplicado > subtotalItem Then
        descontoAplicado = subtotalItem
        MsgBox "Desconto ajustado para o valor máximo possível: " & Format(descontoAplicado, "R$ #,##0.00"), vbInformation
    End If
    
    novoTotal = subtotalItem - descontoAplicado
    
    Me.lstSelecionados.List(indiceItem, 5) = Format(descontoAplicado, "R$ #,##0.00")
    Me.lstSelecionados.List(indiceItem, 6) = Format(novoTotal, "R$ #,##0.00")
    
    Call CalcularTodosOsTotais
    
    MsgBox "Desconto de " & Format(valorDesconto, "R$ #,##0.00") & " aplicado!" & vbCrLf & _
           "Novo Total: " & Format(novoTotal, "R$ #,##0.00"), vbInformation
    
    On Error GoTo 0
End Sub

' === FUNÇÃO PARA REMOVER DESCONTO ===
Private Sub btnRemoverDesconto_Click()
    On Error Resume Next
    
    If Me.lstSelecionados.ListIndex < 0 Then
        MsgBox "Selecione um produto para remover desconto.", vbExclamation
        Exit Sub
    End If
    
    Dim indiceItem As Long
    indiceItem = Me.lstSelecionados.ListIndex
    
    Dim precoUnitario As Currency
    Dim qtdReal As Long
    Dim totalSemDesconto As Currency
    
    precoUnitario = TextoParaMoeda(CStr(Me.lstSelecionados.List(indiceItem, 3)))
    qtdReal = QuantidadePDVParaReal(CStr(Me.lstSelecionados.List(indiceItem, 4)))

    Call CalcularTodosOsTotais
    MsgBox "Desconto removido com sucesso!", vbInformation
    
    On Error GoTo 0
End Sub

' === FUNÇÃO CENTRAL: CALCULA TODOS OS TOTAIS ===
Private Sub CalcularTodosOsTotais()
    On Error Resume Next
    
    Dim i As Long
    Dim precoUnit As Currency
    Dim qtdReal As Long
    Dim desconto As Currency
    Dim valorFinalItem As Currency
    Dim subtotalItem As Currency
    
    totalSubtotal = 0
    totalDescontos = 0
    totalFinal = 0
    quantidadeItens = Me.lstSelecionados.ListCount
    
    For i = 0 To quantidadeItens - 1
        precoUnit = TextoParaMoeda(CStr(Me.lstSelecionados.List(i, 3)))
        qtdReal = QuantidadePDVParaReal(CStr(Me.lstSelecionados.List(i, 4)))
        desconto = TextoParaMoeda(CStr(Me.lstSelecionados.List(i, 5)))
        valorFinalItem = TextoParaMoeda(CStr(Me.lstSelecionados.List(i, 6)))
        
        subtotalItem = precoUnit * qtdReal
        
        totalSubtotal = totalSubtotal + subtotalItem
        totalDescontos = totalDescontos + desconto
        totalFinal = totalFinal + valorFinalItem
    Next i
    
    Call AtualizarLabelsDoTotais
    
    On Error GoTo 0
End Sub

' === ATUALIZAÇÃO DOS LABELS COM VERIFICAÇÃO DE EXISTÊNCIA ===
Private Sub AtualizarLabelsDoTotais()
    On Error Resume Next
    
    If Not Me.Controls("lblSubTotal") Is Nothing Then
        Me.lblSubTotal.Caption = "Subtotal: " & Format(totalSubtotal, "R$ #,##0.00")
    End If
    
    If Not Me.Controls("lblTotal") Is Nothing Then
        Me.lblTotal.Caption = "Total: " & Format(totalFinal, "R$ #,##0.00")
    End If
    
    If Not Me.Controls("lblTotalItens") Is Nothing Then
        Me.lblTotalItens.Caption = "Itens: " & quantidadeItens
    End If
    
    If Not Me.Controls("lblDesc") Is Nothing Then
        If totalDescontos > 0 Then
            Me.lblDesc.Caption = "Desconto Total: " & Format(totalDescontos, "R$ #,##0.00")
        Else
            Me.lblDesc.Caption = "Sem desconto aplicado"
        End If
    End If
    
    On Error GoTo 0
End Sub

' === FUNÇÃO AUXILIAR: CONTAR PRODUTOS REAIS ===
Private Function ObterTotalProdutosReais() As Long
    On Error Resume Next
    
    Dim i As Long, totalReal As Long
    totalReal = 0
    
    For i = 0 To Me.lstSelecionados.ListCount - 1
        totalReal = totalReal + QuantidadePDVParaReal(CStr(Me.lstSelecionados.List(i, 4)))

    ObterTotalProdutosReais = totalReal
    On Error GoTo 0
End Function

' === PESQUISA DE PRODUTOS ===
Private Sub txtPesquisa_Change()
    On Error Resume Next
    Dim termoBusca As String
    termoBusca = UCase(Trim(Me.txtPesquisa.Value))
    If Len(termoBusca) >= 2 Then
        Call PesquisarProdutos(termoBusca)
    ElseIf Len(termoBusca) = 0 Then
        Call CarregarTodosProdutos
    End If
    On Error GoTo 0
End Sub

Private Sub btnLimparPesquisa_Click()
    On Error Resume Next
    Me.txtPesquisa.Value = ""
    Me.txtPesquisa.SetFocus
    Call txtPesquisa_Change
    On Error GoTo 0
End Sub

' === CARREGAMENTO DE PRODUTOS ===
Private Sub CarregarTodosProdutos()
    On Error GoTo TratarErro
    
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Worksheets("Produtos")
    Dim ultimaLinha As Long
    ultimaLinha = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    
    Me.lstProdutos.Clear
    
    If ultimaLinha < 2 Then
        MsgBox "Nenhum produto cadastrado na planilha!", vbExclamation
        Exit Sub
    End If
    
    Dim i As Long
    For i = 2 To ultimaLinha
        If Trim(ws.Cells(i, 1).Value) <> "" Then
            With Me.lstProdutos
                .AddItem
                .List(.ListCount - 1, 0) = ws.Cells(i, 1).Value
                .List(.ListCount - 1, 1) = ws.Cells(i, 2).Value
                .List(.ListCount - 1, 2) = ws.Cells(i, 3).Value
                .List(.ListCount - 1, 3) = ws.Cells(i, 4).Value
                .List(.ListCount - 1, 4) = Format(ws.Cells(i, 6).Value, "R$ #,##0.00")
            End With
        End If
    Next i
    
    Exit Sub
    
TratarErro:
    MsgBox "Erro ao carregar produtos: " & Err.Description, vbCritical
End Sub

' === PESQUISA DE PRODUTOS ===
Private Sub PesquisarProdutos(termo As String)
    On Error GoTo TratarErro
    
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Worksheets("Produtos")
    
    Dim ultimaLinha As Long
    ultimaLinha = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    
    Me.lstProdutos.Clear
    
    If ultimaLinha < 2 Then Exit Sub
    
    Dim i As Long
    For i = 2 To ultimaLinha
        If UCase(ws.Cells(i, 1).Value) Like "*" & UCase(termo) & "*" Or _
           UCase(ws.Cells(i, 2).Value) Like "*" & UCase(termo) & "*" Or _
           UCase(ws.Cells(i, 3).Value) Like "*" & UCase(termo) & "*" Then
            
            With Me.lstProdutos
                .AddItem
                .List(.ListCount - 1, 0) = ws.Cells(i, 1).Value
                .List(.ListCount - 1, 1) = ws.Cells(i, 2).Value
                .List(.ListCount - 1, 2) = ws.Cells(i, 3).Value
                .List(.ListCount - 1, 3) = ws.Cells(i, 4).Value
                .List(.ListCount - 1, 4) = Format(ws.Cells(i, 6).Value, "R$ #,##0.00")
            End With
        End If
    Next i
    
    Exit Sub
    
TratarErro:
    MsgBox "Erro na pesquisa: " & Err.Description, vbCritical
End Sub

' === FUNÇÃO PARA CONVERTER TEXTO DE MOEDA ===
Private Function TextoParaMoeda(texto As String) As Currency
    On Error Resume Next
    
    Dim temp As String
    temp = Trim(CStr(texto))
    
    If temp = "" Or temp = "R$" Then
        TextoParaMoeda = 0
        Exit Function
    End If
    
    temp = Replace(temp, "R$", "")
    temp = Replace(temp, " ", "")
    
    If InStr(temp, ".") > 0 And InStr(temp, ",") > 0 Then
        temp = Replace(temp, ".", "")
        temp = Replace(temp, ",", ".")
    ElseIf InStr(temp, ",") > 0 And Len(temp) - InStr(temp, ",") = 2 Then
        temp = Replace(temp, ",", ".")
    End If
    
    If IsNumeric(temp) Then
        TextoParaMoeda = CCur(temp)
    Else
        TextoParaMoeda = 0
    End If
    
    On Error GoTo 0
End Function

' === ENVIO PARA FORMULÁRIO PRINCIPAL (PROTEGIDO) ===
Private Sub btnEnviarParaProdutos_Click()
    On Error GoTo TratarErro
    
    If Me.lstSelecionados.ListCount = 0 Then
        MsgBox "Adicione produtos antes de enviar.", vbExclamation
        Exit Sub
    End If

    ' Verificar se o formulário principal existe
    Dim formPrincipal As Object
    Dim formEncontrado As Boolean
    formEncontrado = False
    
    On Error Resume Next
    For Each formPrincipal In VBA.UserForms
        If TypeName(formPrincipal) = "frmPDVPrincipal" Then
            formEncontrado = True
            Exit For
        End If
    Next formPrincipal
    On Error GoTo 0

    If Not formEncontrado Then
        MsgBox "Formulário principal não está aberto." & vbCrLf & _
               "Abra primeiro o formulário principal do PDV.", vbExclamation
        Exit Sub
    End If

    ' Tentar encontrar a lista de produtos no formulário principal
    Dim listaDestino As Object
    Dim nomeLista As String
    
    On Error Resume Next
    Set listaDestino = formPrincipal.Controls("produtosv2")
    If Not listaDestino Is Nothing Then
        nomeLista = "produtosv2"
    Else
        Set listaDestino = formPrincipal.Controls("produtosv1")
        If Not listaDestino Is Nothing Then
            nomeLista = "produtosv1"
        Else
            MsgBox "Não foi possível encontrar a lista de produtos no formulário principal.", vbExclamation
            Exit Sub
        End If
    End If
    On Error GoTo 0

    ' Configurar a lista de destino
    With listaDestino
        .ColumnCount = 7
        .ColumnWidths = "60;200;40;80;40;80;80"
    End With

    ' Transferir produtos com verificação de segurança
    Dim i As Long
    For i = 0 To Me.lstSelecionados.ListCount - 1
        With listaDestino
            .AddItem
            Dim novoIndice As Long
            novoIndice = .ListCount - 1
            
            .List(novoIndice, 0) = Me.lstSelecionados.List(i, 0)
            .List(novoIndice, 1) = Me.lstSelecionados.List(i, 1)
            .List(novoIndice, 2) = Me.lstSelecionados.List(i, 2)
            .List(novoIndice, 3) = Me.lstSelecionados.List(i, 3)
            .List(novoIndice, 4) = Me.lstSelecionados.List(i, 4)
            .List(novoIndice, 5) = Me.lstSelecionados.List(i, 5)
            .List(novoIndice, 6) = Me.lstSelecionados.List(i, 6)
        End With
    Next i

    ' Mensagem de sucesso
    MsgBox "Produtos enviados com sucesso para o formulário principal!" & vbCrLf & vbCrLf & _
           "Itens enviados: " & Me.lstSelecionados.ListCount & vbCrLf & _
           "Total: " & Format(totalFinal, "R$ #,##0.00") & vbCrLf & _
           "Lista de destino: " & nomeLista, vbInformation

    ' Limpar lista após envio
    Me.lstSelecionados.Clear
    Call CalcularTodosOsTotais
    
    Exit Sub

TratarErro:
    MsgBox "Erro ao enviar produtos:" & vbCrLf & _
           "Erro: " & Err.Description & vbCrLf & vbCrLf & _
           "Verifique se o formulário principal está aberto e configurado corretamente.", vbCritical
End Sub

' === BOTÃO PARA FECHAR FORMULÁRIO ===
Private Sub btnFechar_Click()
    On Error Resume Next

    ' Formata como "1" (apenas o número) em vez de "1,000"
    FormatarQuantidadePDV = CStr(quantidade)

' === NOVA FUNÇÃO PARA CONVERTER QUANTIDADE PDV ===
Private Function QuantidadePDVParaReal(quantidadePDV As String) As Long
    On Error Resume Next
    Dim temp As String
    temp = Trim(CStr(quantidadePDV))
    
    If temp = "" Then
        QuantidadePDVParaReal = 1
        Exit Function
    End If
    
    ' Remove quaisquer caracteres não numéricos (exceto vírgula e ponto)
    Dim i As Integer
    Dim cleaned As String
    cleaned = ""
    
    For i = 1 To Len(temp)
        Dim char As String
        char = Mid(temp, i, 1)
        If IsNumeric(char) Then
            cleaned = cleaned & char
        ElseIf char = "," Or char = "." Then
            ' Se encontrar decimal, para aqui (não usamos decimais)
            Exit For
        End If
    Next i
    
    If cleaned = "" Then
        QuantidadePDVParaReal = 1
    Else
        QuantidadePDVParaReal = CLng(cleaned)
    End If
    On Error GoTo 0
End Function

Private Function QuantidadePDVParaReal(quantidadePDV As String) As Long
    On Error Resume Next
    Dim temp As String
    temp = Trim(CStr(quantidadePDV))
    
    If temp = "" Then
        QuantidadePDVParaReal = 1
        Exit Function
    End If
    
    ' CORREÇÃO PRINCIPAL: Se terminar com ",000" é formato visual PDV
    ' Remove ",000" e interpreta apenas o número antes da vírgula

        Dim valorNumerico As Double
        valorNumerico = CDbl(temp)
        
        ' Garantir que está dentro dos limites
        If valorNumerico >= QTD_MINIMA And valorNumerico <= QTD_MAXIMA Then
            QuantidadePDVParaReal = CLng(valorNumerico)
        Else
            QuantidadePDVParaReal = 1
        End If
    Else
        QuantidadePDVParaReal = 1
    End If
    
    On Error GoTo 0
End Function

' === FUNÇÃO PARA CONVERTER TEXTO DE MOEDA (CORRIGIDA) ===
Private Function TextoParaMoeda(texto As String) As Currency
    On Error Resume Next
    
    Dim temp As String
    temp = Trim(CStr(texto))
    
    If temp = "" Or temp = "R$" Then
        TextoParaMoeda = 0
        Exit Function
    End If
    
    ' Remover símbolo da moeda e espaços
    temp = Replace(temp, "R$", "")
    temp = Replace(temp, " ", "")
    temp = Trim(temp)
    
    ' Corrigir formatação de moeda brasileira
    If InStr(temp, ".") > 0 And InStr(temp, ",") > 0 Then
        ' Formato: 1.234,56
        temp = Replace(temp, ".", "")
        temp = Replace(temp, ",", ".")
    ElseIf InStr(temp, ",") > 0 And Len(temp) - InStr(temp, ",") = 2 Then
        ' Formato: 123,45
        temp = Replace(temp, ",", ".")
    End If
    
    If IsNumeric(temp) Then
        TextoParaMoeda = CCur(temp)
    Else
        TextoParaMoeda = 0
    End If
    
    On Error GoTo 0
End Function

' === FUNCAO PARA APLICAR DESCONTO EM ITEM ===
Private Sub AplicarDescontoItem(indiceItem As Long, valorDesconto As Double)
    On Error GoTo TratarErro
    
    Dim precoUnitario As Currency
    Dim qtdReal As Long
    Dim descontoAplicado As Currency
    Dim subtotalItem As Currency
    Dim novoTotal As Currency
    
    precoUnitario = TextoParaMoeda(CStr(Me.lstSelecionados.List(indiceItem, 3)))
    qtdReal = QuantidadePDVParaReal(CStr(Me.lstSelecionados.List(indiceItem, 4)))
    subtotalItem = precoUnitario * qtdReal
    
    descontoAplicado = valorDesconto
    
    If descontoAplicado > subtotalItem Then
        descontoAplicado = subtotalItem
    End If
    
    novoTotal = subtotalItem - descontoAplicado
    
    Me.lstSelecionados.List(indiceItem, 5) = Format(descontoAplicado, "R$ #,##0.00")
    Me.lstSelecionados.List(indiceItem, 6) = Format(novoTotal, "R$ #,##0.00")
    
    Call CalcularTodosOsTotais
    
    MsgBox "Desconto de " & Format(valorDesconto, "R$ #,##0.00") & " aplicado!" & vbCrLf & _
           "Novo Total: " & Format(novoTotal, "R$ #,##0.00"), vbInformation
    
    Exit Sub
TratarErro:
    MsgBox "Erro ao aplicar desconto: " & Err.Description, vbCritical
End Sub

' === FUNCAO CENTRAL: CALCULA TODOS OS TOTAIS (CORRIGIDA) ===
Private Sub CalcularTodosOsTotais()
    On Error Resume Next
    
    Dim i As Long
    Dim precoUnit As Currency
    Dim qtdReal As Long
    Dim desconto As Currency
    Dim valorFinalItem As Currency
    Dim subtotalItem As Currency
    
    totalSubtotal = 0
    totalDescontos = 0
    totalFinal = 0
    quantidadeItens = Me.lstSelecionados.ListCount
    
    For i = 0 To quantidadeItens - 1
        precoUnit = TextoParaMoeda(CStr(Me.lstSelecionados.List(i, 3)))
        qtdReal = QuantidadePDVParaReal(CStr(Me.lstSelecionados.List(i, 4)))
        desconto = TextoParaMoeda(CStr(Me.lstSelecionados.List(i, 5)))
        
        ' CORREÇÃO PRINCIPAL: Calcular subtotal baseado na quantidade real
        subtotalItem = precoUnit * qtdReal
        valorFinalItem = subtotalItem - desconto
        
        totalSubtotal = totalSubtotal + subtotalItem
        totalDescontos = totalDescontos + desconto
        totalFinal = totalFinal + valorFinalItem
        
        ' DEBUG: Mostrar cálculos no console
        Debug.Print "Item " & i & ": Preço=" & precoUnit & " x Qtd=" & qtdReal & " = " & subtotalItem
    Next i
    
    Call AtualizarLabelsDoTotais
    On Error GoTo 0
End Sub

' === ATUALIZACAO DOS LABELS COM VERIFICACAO DE EXISTENCIA ===
Private Sub AtualizarLabelsDoTotais()
    On Error Resume Next
    
    ' Atualizar lblSubTotal (mantém o nome original do usuário)
    If Not Me.Controls("lblSubTotal") Is Nothing Then
        Me.lblSubTotal.Caption = Format(totalSubtotal, "R$ #,##0.00")
    End If
    
    ' Atualizar lblValorTotal (mantém o nome original do usuário) 
    If Not Me.Controls("lblValorTotal") Is Nothing Then
        Me.lblValorTotal.Caption = Format(totalFinal, "R$ #,##0.00")
    End If
    
    ' Verificar outros possíveis nomes de labels
    If Not Me.Controls("lblTotal") Is Nothing Then
        Me.lblTotal.Caption = "Total: " & Format(totalFinal, "R$ #,##0.00")
    End If
    
    If Not Me.Controls("lblTotalItens") Is Nothing Then
        Me.lblTotalItens.Caption = "Itens: " & quantidadeItens
    End If
    
    If Not Me.Controls("lblDesc") Is Nothing Then
        If totalDescontos > 0 Then
            Me.lblDesc.Caption = "Desconto Total: " & Format(totalDescontos, "R$ #,##0.00")
        Else
            Me.lblDesc.Caption = "Sem desconto aplicado"
        End If
    End If
    
    On Error GoTo 0
    
    Debug.Print "=== TOTAIS PDV MADEIREIRA (CORRIGIDOS) ==="
    Debug.Print "Itens na lista: " & quantidadeItens
    Debug.Print "Produtos reais: " & ObterTotalProdutosReais()
    Debug.Print "Subtotal: " & Format(totalSubtotal, "R$ #,##0.00")
    Debug.Print "Descontos: " & Format(totalDescontos, "R$ #,##0.00")
    Debug.Print "Total Final: " & Format(totalFinal, "R$ #,##0.00")
    Debug.Print "============================================"
End Sub

' === FUNCAO AUXILIAR: CONTAR PRODUTOS REAIS ===

' === CARREGAMENTO DE PRODUTOS ===
Private Sub CarregarTodosProdutos()
    On Error GoTo TratarErro
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Worksheets("Produtos")
    Dim ultimaLinha As Long
    ultimaLinha = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    
    Me.lstProdutos.Clear
    
    Dim i As Long
    For i = 2 To ultimaLinha
        If Trim(CStr(ws.Cells(i, 1).Value)) <> "" Then
            With Me.lstProdutos
                .AddItem
                .List(.ListCount - 1, 0) = ws.Cells(i, 1).Value
                .List(.ListCount - 1, 1) = ws.Cells(i, 2).Value
                .List(.ListCount - 1, 2) = ws.Cells(i, 3).Value
                .List(.ListCount - 1, 3) = ws.Cells(i, 4).Value
                .List(.ListCount - 1, 4) = Format(ws.Cells(i, 6).Value, "R$ #,##0.00")
            End With
        End If
    Next i
    Exit Sub
TratarErro:
    MsgBox "Erro ao carregar produtos: " & Err.Description, vbCritical
End Sub

' === PESQUISA DE PRODUTOS ===
Private Sub PesquisarProdutos(termo As String)
    On Error GoTo TratarErro
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Worksheets("Produtos")
    Me.lstProdutos.Clear
    
    Dim i As Long
    For i = 2 To ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
        If UCase(CStr(ws.Cells(i, 1).Value)) Like "*" & termo & "*" Or _
           UCase(CStr(ws.Cells(i, 2).Value)) Like "*" & termo & "*" Then
            With Me.lstProdutos
                .AddItem
                .List(.ListCount - 1, 0) = ws.Cells(i, 1).Value
                .List(.ListCount - 1, 1) = ws.Cells(i, 2).Value
                .List(.ListCount - 1, 2) = ws.Cells(i, 3).Value
                .List(.ListCount - 1, 3) = ws.Cells(i, 4).Value
                .List(.ListCount - 1, 4) = Format(ws.Cells(i, 6).Value, "R$ #,##0.00")
            End With
        End If
    Next i
    Exit Sub
TratarErro:
    MsgBox "Erro na pesquisa: " & Err.Description, vbCritical
End Sub

' === ENVIO PARA FORMULARIO PRINCIPAL ===
Private Sub btnEnviarParaProdutos_Click()
    On Error GoTo ErroEnvio
    
    Dim formPDV As Object
    Dim frm As Object
    Dim formEncontrado As Boolean
    Dim listaDestino As Object
    Dim nomeControle As String
    Dim i As Long, n As Long
    
    If Me.lstSelecionados.ListCount = 0 Then
        MsgBox "Adicione produtos antes de enviar.", vbExclamation
        Exit Sub
    End If

    formEncontrado = False

    For Each frm In VBA.UserForms
        Debug.Print "Formulário encontrado: " & TypeName(frm)
        If TypeName(frm) = "frmPDVPrincipal" Then
            Set formPDV = frm
            formEncontrado = True
            Debug.Print "Formulário frmPDVPrincipal encontrado!"
            Exit For
        End If
    Next frm

    If Not formEncontrado Or formPDV Is Nothing Then
        MsgBox "O formulário frmPDVPrincipal não está aberto." & vbCrLf & _
               "Abra primeiro o formulário principal.", vbCritical
        Exit Sub
    End If

    On Error Resume Next
    Set listaDestino = formPDV.Controls("produtosv2")
    If Not listaDestino Is Nothing Then
        nomeControle = "produtosv2"
        Debug.Print "Lista produtosv2 encontrada!"
    Else
        Set listaDestino = formPDV.Controls("produtosv1")
        If Not listaDestino Is Nothing Then
            nomeControle = "produtosv1"
            Debug.Print "Lista produtosv1 encontrada!"
        End If
    End If
    On Error GoTo ErroEnvio

    If listaDestino Is Nothing Then
        MsgBox "Lista de produtos não encontrada no formulário principal." & vbCrLf & _
               "Verifique se existe 'produtosv1' ou 'produtosv2'.", vbCritical
        Exit Sub
    End If

    With listaDestino
        .ColumnCount = 7
        .ColumnWidths = "60;200;40;80;40;80;80"
        Debug.Print "Lista " & nomeControle & " configurada com " & .ColumnCount & " colunas"
        Debug.Print "Lista " & nomeControle & " tem " & .ListCount & " itens existentes"
        Debug.Print "Vai adicionar mais " & Me.lstSelecionados.ListCount & " itens"
    End With
    
    For i = 0 To Me.lstSelecionados.ListCount - 1
        With listaDestino
            .AddItem Me.lstSelecionados.List(i, 0)
            n = .ListCount - 1
            
            .List(n, 1) = Me.lstSelecionados.List(i, 1)
            .List(n, 2) = Me.lstSelecionados.List(i, 2)
            .List(n, 3) = Me.lstSelecionados.List(i, 3)
            .List(n, 4) = Me.lstSelecionados.List(i, 4)
            .List(n, 5) = Me.lstSelecionados.List(i, 5)
            .List(n, 6) = Me.lstSelecionados.List(i, 6)
        End With
    Next i

    Debug.Print "Envio concluído!"
    Debug.Print "Total de itens enviados: " & Me.lstSelecionados.ListCount
    Debug.Print "Lista de destino: " & nomeControle
    Debug.Print "Total final enviado: " & Format(totalFinal, "R$ #,##0.00")
    
    With listaDestino
        Debug.Print "Itens na lista destino: " & .ListCount
        If .ListCount > 0 Then
            Debug.Print "Último item enviado:"
            Debug.Print "  Col 0: '" & .List(.ListCount - 1, 0) & "'"
            Debug.Print "  Col 1: '" & .List(.ListCount - 1, 1) & "'"
            Debug.Print "  Col 6: '" & .List(.ListCount - 1, 6) & "'"
        End If
    End With
    
    MsgBox "PRODUTOS ENVIADOS!" & vbCrLf & vbCrLf & _
           "Itens: " & Me.lstSelecionados.ListCount & vbCrLf & _
           "Total: " & Format(totalFinal, "R$ #,##0.00") & vbCrLf & _
           "Para: " & nomeControle & vbCrLf & vbCrLf & _
           "Lista agora tem " & listaDestino.ListCount & " itens total", vbInformation
    
    Me.lstSelecionados.Clear
    Call CalcularTodosOsTotais
    Unload Me
    Exit Sub

ErroEnvio:
    MsgBox "Erro ao enviar produtos:" & vbCrLf & _
           "Linha: " & Erl & vbCrLf & _
           "Erro: " & Err.Description & vbCrLf & vbCrLf & _
           "Tente novamente.", vbCritical
End Sub

        MsgBox "Quantidade não foi alterada.", vbInformation
        Exit Sub
    End If
    
    Dim qtdDisplayNova As String
    Dim novoSubtotal As Currency
    Dim novoTotal As Currency
    
    qtdDisplayNova = FormatarQuantidadePDV(qtdRealNova)
    novoSubtotal = precoUnitario * qtdRealNova
    
    ' Ajustar desconto se necessário
    If descontoAtual > novoSubtotal Then
        descontoAtual = novoSubtotal
    End If
    
    novoTotal = novoSubtotal - descontoAtual
    
    Dim confirmacao As VbMsgBoxResult
    confirmacao = MsgBox( _
        "CONFIRMAR ALTERAÇÃO" & vbCrLf & vbCrLf & _
        nomeItem & vbCrLf & vbCrLf & _
        "Quantidade: " & qtdRealAtual & " -> " & qtdRealNova & vbCrLf & _
        "Display: " & qtdDisplayAtual & " -> " & qtdDisplayNova & vbCrLf & _
        "Subtotal: " & Format(precoUnitario * qtdRealAtual, "R$ #,##0.00") & " -> " & Format(novoSubtotal, "R$ #,##0.00") & vbCrLf & _
        "Total: " & Format(novoTotal, "R$ #,##0.00") & vbCrLf & vbCrLf & _
        "Confirma?", vbYesNo + vbQuestion, "Confirmar Alteração")
    
    If confirmacao = vbNo Then Exit Sub
    
    ' Atualizar os dados na lista
    Me.lstSelecionados.List(indiceItem, 4) = qtdDisplayNova
    Me.lstSelecionados.List(indiceItem, 5) = Format(descontoAtual, "R$ #,##0.00")
    Me.lstSelecionados.List(indiceItem, 6) = Format(novoTotal, "R$ #,##0.00")
    
    Call CalcularTodosOsTotais
    
    MsgBox "QUANTIDADE ALTERADA!" & vbCrLf & vbCrLf & _
           nomeItem & vbCrLf & _
           "Quantidade: " & qtdRealNova & " produto(s)" & vbCrLf & _
           "Display: " & qtdDisplayNova & " UN" & vbCrLf & _
           "Total: " & Format(novoTotal, "R$ #,##0.00"), vbInformation
    
    Exit Sub
TratarErro:
    MsgBox "Erro ao alterar quantidade: " & Err.Description, vbCritical
End Sub

' === BOTAO ADICIONAR MODIFICADO PARA PDV MADEIREIRA ===
Private Sub btnAdicionar_Click()
    On Error GoTo TratarErro
    
    If Me.lstProdutos.ListIndex < 0 Then
        MsgBox "Selecione um produto.", vbExclamation
        Exit Sub
    End If
    
    Dim i As Long, idProduto As String
    idProduto = CStr(Me.lstProdutos.List(Me.lstProdutos.ListIndex, 0))
    
    ' Verificar se produto já foi adicionado
    For i = 0 To Me.lstSelecionados.ListCount - 1
        If CStr(Me.lstSelecionados.List(i, 0)) = idProduto Then
            MsgBox "Produto já adicionado.", vbInformation
            Exit Sub
        End If
    Next i
    
    ' Buscar preço do produto na planilha
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Worksheets("Produtos")
    
    Dim precoUnitario As Currency
    Dim linhaAtual As Long
    
    For linhaAtual = 2 To ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
        If CStr(ws.Cells(linhaAtual, 1).Value) = idProduto Then
            precoUnitario = ws.Cells(linhaAtual, 6).Value
            Exit For
        End If
    Next linhaAtual
    
    ' Obter quantidade
    Dim qtdReal As Long
    qtdReal = CLng(Me.txtQuantidade.Value)
    
    If qtdReal < QTD_MINIMA Or qtdReal > QTD_MAXIMA Then
        MsgBox "Quantidade deve estar entre " & QTD_MINIMA & " e " & QTD_MAXIMA, vbExclamation
        Exit Sub
    End If
    
    Dim qtdDisplay As String
    Dim subtotalItem As Currency
    
    qtdDisplay = FormatarQuantidadePDV(qtdReal)
    subtotalItem = precoUnitario * qtdReal
    
    ' Adicionar item na lista
    With Me.lstSelecionados
        .AddItem
        .List(.ListCount - 1, 0) = idProduto
        .List(.ListCount - 1, 1) = Me.lstProdutos.List(Me.lstProdutos.ListIndex, 1)
        .List(.ListCount - 1, 2) = "UN"
        .List(.ListCount - 1, 3) = Format(precoUnitario, "R$ #,##0.00")
        .List(.ListCount - 1, 4) = qtdDisplay
        .List(.ListCount - 1, 5) = "R$ 0,00"
        .List(.ListCount - 1, 6) = Format(subtotalItem, "R$ #,##0.00")
    End With
    
    Call CalcularTodosOsTotais
    
    ' Resetar quantidade
    Me.txtQuantidade.Value = QTD_MINIMA
    
    MsgBox "Produto adicionado!" & vbCrLf & vbCrLf & _
           "Quantidade: " & qtdDisplay & " UN (" & qtdReal & " produtos)" & vbCrLf & _
           "Preço: " & Format(precoUnitario, "R$ #,##0.00") & " cada" & vbCrLf & _
           "Total: " & Format(subtotalItem, "R$ #,##0.00"), vbInformation
    
    Exit Sub
TratarErro:
    MsgBox "Erro ao adicionar produto: " & Err.Description, vbCritical
End Sub

' === FUNCAO PARA APLICAR DESCONTO EM ITEM ===
Private Sub AplicarDescontoItem(indiceItem As Long, valorDesconto As Double)
    On Error GoTo TratarErro
    
    Dim precoUnitario As Currency
    Dim qtdReal As Long
    Dim descontoAplicado As Currency
    Dim subtotalItem As Currency
    Dim novoTotal As Currency
    
    precoUnitario = TextoParaMoeda(CStr(Me.lstSelecionados.List(indiceItem, 3)))
    qtdReal = QuantidadePDVParaReal(CStr(Me.lstSelecionados.List(indiceItem, 4)))
    subtotalItem = precoUnitario * qtdReal
    
    descontoAplicado = valorDesconto
    
    If descontoAplicado > subtotalItem Then
        descontoAplicado = subtotalItem
    End If
    
    novoTotal = subtotalItem - descontoAplicado
    
    Me.lstSelecionados.List(indiceItem, 5) = Format(descontoAplicado, "R$ #,##0.00")
    Me.lstSelecionados.List(indiceItem, 6) = Format(novoTotal, "R$ #,##0.00")
    
    Call CalcularTodosOsTotais
    
    MsgBox "Desconto de " & Format(valorDesconto, "R$ #,##0.00") & " aplicado!" & vbCrLf & _
           "Novo Total: " & Format(novoTotal, "R$ #,##0.00"), vbInformation
    
    Exit Sub
TratarErro:
    MsgBox "Erro ao aplicar desconto: " & Err.Description, vbCritical
End Sub

' === FUNCAO CENTRAL: CALCULA TODOS OS TOTAIS (CORRIGIDA) ===
Private Sub CalcularTodosOsTotais()
    On Error Resume Next
    
    Dim i As Long
    Dim precoUnit As Currency
    Dim qtdReal As Long
    Dim desconto As Currency
    Dim valorFinalItem As Currency
    Dim subtotalItem As Currency
    
    totalSubtotal = 0
    totalDescontos = 0
    totalFinal = 0
    quantidadeItens = Me.lstSelecionados.ListCount
    
    For i = 0 To quantidadeItens - 1
        precoUnit = TextoParaMoeda(CStr(Me.lstSelecionados.List(i, 3)))
        qtdReal = QuantidadePDVParaReal(CStr(Me.lstSelecionados.List(i, 4)))
        desconto = TextoParaMoeda(CStr(Me.lstSelecionados.List(i, 5)))
        
        ' CORREÇÃO PRINCIPAL: Calcular subtotal baseado na quantidade real
        subtotalItem = precoUnit * qtdReal
        valorFinalItem = subtotalItem - desconto
        
        totalSubtotal = totalSubtotal + subtotalItem
        totalDescontos = totalDescontos + desconto
        totalFinal = totalFinal + valorFinalItem
        
        ' DEBUG: Mostrar cálculos no console
        Debug.Print "Item " & i & ": Preço=" & precoUnit & " x Qtd=" & qtdReal & " = " & subtotalItem
    Next i
    
    Call AtualizarLabelsDoTotais
    On Error GoTo 0
End Sub

' === ATUALIZACAO DOS LABELS COM VERIFICACAO DE EXISTENCIA ===
Private Sub AtualizarLabelsDoTotais()
    On Error Resume Next
    
    ' Atualizar lblSubTotal (mantém o nome original do usuário)
    If Not Me.Controls("lblSubTotal") Is Nothing Then
        Me.lblSubTotal.Caption = Format(totalSubtotal, "R$ #,##0.00")
    End If
    
    ' Atualizar lblValorTotal (mantém o nome original do usuário) 
    If Not Me.Controls("lblValorTotal") Is Nothing Then
        Me.lblValorTotal.Caption = Format(totalFinal, "R$ #,##0.00")
    End If
    
    ' Verificar outros possíveis nomes de labels
    If Not Me.Controls("lblTotal") Is Nothing Then
        Me.lblTotal.Caption = "Total: " & Format(totalFinal, "R$ #,##0.00")
    End If
    
    If Not Me.Controls("lblTotalItens") Is Nothing Then
        Me.lblTotalItens.Caption = "Itens: " & quantidadeItens
    End If
    
    If Not Me.Controls("lblDesc") Is Nothing Then
        If totalDescontos > 0 Then
            Me.lblDesc.Caption = "Desconto Total: " & Format(totalDescontos, "R$ #,##0.00")
        Else
            Me.lblDesc.Caption = "Sem desconto aplicado"
        End If
    End If
    
    On Error GoTo 0
    
    Debug.Print "=== TOTAIS PDV MADEIREIRA (CORRIGIDOS) ==="
    Debug.Print "Itens na lista: " & quantidadeItens
    Debug.Print "Produtos reais: " & ObterTotalProdutosReais()
    Debug.Print "Subtotal: " & Format(totalSubtotal, "R$ #,##0.00")
    Debug.Print "Descontos: " & Format(totalDescontos, "R$ #,##0.00")
    Debug.Print "Total Final: " & Format(totalFinal, "R$ #,##0.00")
    Debug.Print "============================================"
End Sub

' === ENVIO PARA FORMULARIO PRINCIPAL ===
Private Sub btnEnviarParaProdutos_Click()
    On Error GoTo ErroEnvio
    
    Dim formPDV As Object
    Dim frm As Object
    Dim formEncontrado As Boolean
    Dim listaDestino As Object
    Dim nomeControle As String
    Dim i As Long, n As Long
    
    If Me.lstSelecionados.ListCount = 0 Then
        MsgBox "Adicione produtos antes de enviar.", vbExclamation
        Exit Sub
    End If

    formEncontrado = False

    For Each frm In VBA.UserForms
        Debug.Print "Formulário encontrado: " & TypeName(frm)
        If TypeName(frm) = "frmPDVPrincipal" Then
            Set formPDV = frm
            formEncontrado = True
            Debug.Print "Formulário frmPDVPrincipal encontrado!"
            Exit For
        End If
    Next frm

    If Not formEncontrado Or formPDV Is Nothing Then
        MsgBox "O formulário frmPDVPrincipal não está aberto." & vbCrLf & _
               "Abra primeiro o formulário principal.", vbCritical
        Exit Sub
    End If

    On Error Resume Next
    Set listaDestino = formPDV.Controls("produtosv2")
    If Not listaDestino Is Nothing Then
        nomeControle = "produtosv2"
        Debug.Print "Lista produtosv2 encontrada!"
    Else
        Set listaDestino = formPDV.Controls("produtosv1")
        If Not listaDestino Is Nothing Then
            nomeControle = "produtosv1"
            Debug.Print "Lista produtosv1 encontrada!"
        End If
    End If
    On Error GoTo ErroEnvio

    If listaDestino Is Nothing Then
        MsgBox "Lista de produtos não encontrada no formulário principal." & vbCrLf & _
               "Verifique se existe 'produtosv1' ou 'produtosv2'.", vbCritical
        Exit Sub
    End If

    With listaDestino
        .ColumnCount = 7
        .ColumnWidths = "60;200;40;80;40;80;80"
        Debug.Print "Lista " & nomeControle & " configurada com " & .ColumnCount & " colunas"
        Debug.Print "Lista " & nomeControle & " tem " & .ListCount & " itens existentes"
        Debug.Print "Vai adicionar mais " & Me.lstSelecionados.ListCount & " itens"
    End With
    
    For i = 0 To Me.lstSelecionados.ListCount - 1
        With listaDestino
            .AddItem Me.lstSelecionados.List(i, 0)
            n = .ListCount - 1
            
            .List(n, 1) = Me.lstSelecionados.List(i, 1)
            .List(n, 2) = Me.lstSelecionados.List(i, 2)
            .List(n, 3) = Me.lstSelecionados.List(i, 3)
            .List(n, 4) = Me.lstSelecionados.List(i, 4)
            .List(n, 5) = Me.lstSelecionados.List(i, 5)
            .List(n, 6) = Me.lstSelecionados.List(i, 6)
        End With
    Next i

    Debug.Print "Envio concluído!"
    Debug.Print "Total de itens enviados: " & Me.lstSelecionados.ListCount
    Debug.Print "Lista de destino: " & nomeControle
    Debug.Print "Total final enviado: " & Format(totalFinal, "R$ #,##0.00")
    
    With listaDestino
        Debug.Print "Itens na lista destino: " & .ListCount
        If .ListCount > 0 Then
            Debug.Print "Último item enviado:"
            Debug.Print "  Col 0: '" & .List(.ListCount - 1, 0) & "'"
            Debug.Print "  Col 1: '" & .List(.ListCount - 1, 1) & "'"
            Debug.Print "  Col 6: '" & .List(.ListCount - 1, 6) & "'"
        End If
    End With
    
    MsgBox "PRODUTOS ENVIADOS!" & vbCrLf & vbCrLf & _
           "Itens: " & Me.lstSelecionados.ListCount & vbCrLf & _
           "Total: " & Format(totalFinal, "R$ #,##0.00") & vbCrLf & _
           "Para: " & nomeControle & vbCrLf & vbCrLf & _
           "Lista agora tem " & listaDestino.ListCount & " itens total", vbInformation
    
    Me.lstSelecionados.Clear
    Call CalcularTodosOsTotais
    Unload Me
    Exit Sub

ErroEnvio:
    MsgBox "Erro ao enviar produtos:" & vbCrLf & _
           "Linha: " & Erl & vbCrLf & _
           "Erro: " & Err.Description & vbCrLf & vbCrLf & _
           "Tente novamente.", vbCritical
End Sub

Private Function FormatarQuantidadePDV(quantidade As Long) As String
    FormatarQuantidadePDV = Format(quantidade, "0") & ",000"
End Function

' === FUNÇÃO PARA CONVERTER QUANTIDADE PDV PARA REAL (CORRIGIDA) ===
Private Function QuantidadePDVParaReal(quantidadePDV As String) As Long
    On Error Resume Next
    Dim temp As String
    temp = Trim(CStr(quantidadePDV))
    
    If temp = "" Then
        QuantidadePDVParaReal = 1
        Exit Function
    End If
    
    ' CORREÇÃO PRINCIPAL: Se terminar com ",000" é formato visual PDV
    If Right(temp, 4) = ",000" Then
        temp = Left(temp, Len(temp) - 4)
    End If
    
    temp = Trim(temp)
    
    If IsNumeric(temp) And temp <> "" Then
        QuantidadePDVParaReal = CLng(temp)
    Else
        QuantidadePDVParaReal = 1
    End If
    On Error GoTo 0
End Function

' === FUNÇÃO PARA CONVERTER TEXTO DE MOEDA (CORRIGIDA) ===
Private Function TextoParaMoeda(texto As String) As Currency
    On Error Resume Next
    
    Dim temp As String
    temp = Trim(CStr(texto))
    
    If temp = "" Or temp = "R$" Then
        TextoParaMoeda = 0
        Exit Function
    End If
    
    temp = Replace(temp, "R$", "")
    temp = Replace(temp, " ", "")
    temp = Trim(temp)
    
    ' Corrigir formatação de moeda brasileira
    If InStr(temp, ".") > 0 And InStr(temp, ",") > 0 Then
        temp = Replace(temp, ".", "")
        temp = Replace(temp, ",", ".")
    ElseIf InStr(temp, ",") > 0 Then
        temp = Replace(temp, ",", ".")
    End If
    
    If IsNumeric(temp) Then
        TextoParaMoeda = CCur(temp)
    Else
        TextoParaMoeda = 0
    End If
    
    On Error GoTo 0
End Function

' === FUNÇÃO PARA EDITAR QUANTIDADE DE ITEM ===
Private Sub EditarQuantidadeItem(indiceItem As Long)
    On Error GoTo TratarErro
    
    Dim nomeItem As String
    Dim qtdDisplayAtual As String
    Dim qtdRealAtual As Long
    Dim precoUnitario As Currency
    Dim descontoAtual As Currency
    
    nomeItem = CStr(Me.lstSelecionados.List(indiceItem, 1))
    qtdDisplayAtual = CStr(Me.lstSelecionados.List(indiceItem, 4))
    qtdRealAtual = QuantidadePDVParaReal(qtdDisplayAtual)
    precoUnitario = TextoParaMoeda(CStr(Me.lstSelecionados.List(indiceItem, 3)))
    descontoAtual = TextoParaMoeda(CStr(Me.lstSelecionados.List(indiceItem, 5)))
    
    Dim novaQtdStr As String
    Dim qtdRealNova As Long
    
    novaQtdStr = InputBox( _
        "ALTERAR QUANTIDADE - SISTEMA PDV MADEIREIRA" & vbCrLf & vbCrLf & _
        "Produto: " & nomeItem & vbCrLf & _
        "Preço: " & Format(precoUnitario, "R$ #,##0.00") & " cada" & vbCrLf & _
        "Quantidade Atual: " & qtdRealAtual & " produto(s)" & vbCrLf & _
        "Display PDV: " & qtdDisplayAtual & " UN" & vbCrLf & vbCrLf & _
        "Digite a nova quantidade:", _
        "Alterar Quantidade", _
        qtdRealAtual)
    
    If novaQtdStr = "" Then Exit Sub
    
    If Not IsNumeric(novaQtdStr) Then
        MsgBox "Digite apenas números inteiros!", vbExclamation
        Exit Sub
    End If
    
    qtdRealNova = CLng(novaQtdStr)
    
    If qtdRealNova < QTD_MINIMA Or qtdRealNova > QTD_MAXIMA Then
        MsgBox "Quantidade deve estar entre " & QTD_MINIMA & " e " & QTD_MAXIMA, vbExclamation
        Exit Sub
    End If
    
    If qtdRealNova = qtdRealAtual Then
        MsgBox "Quantidade não foi alterada.", vbInformation
        Exit Sub
    End If
    
    Dim qtdDisplayNova As String
    Dim novoSubtotal As Currency
    Dim novoTotal As Currency
    
    qtdDisplayNova = FormatarQuantidadePDV(qtdRealNova)
    novoSubtotal = precoUnitario * qtdRealNova
    
    ' Ajustar desconto se necessário
    If descontoAtual > novoSubtotal Then
        descontoAtual = novoSubtotal
    End If
    
    novoTotal = novoSubtotal - descontoAtual
    
    Dim confirmacao As VbMsgBoxResult
    confirmacao = MsgBox( _
        "CONFIRMAR ALTERAÇÃO" & vbCrLf & vbCrLf & _
        nomeItem & vbCrLf & vbCrLf & _
        "Quantidade: " & qtdRealAtual & " -> " & qtdRealNova & vbCrLf & _
        "Display: " & qtdDisplayAtual & " -> " & qtdDisplayNova & vbCrLf & _
        "Subtotal: " & Format(precoUnitario * qtdRealAtual, "R$ #,##0.00") & " -> " & Format(novoSubtotal, "R$ #,##0.00") & vbCrLf & _
        "Total: " & Format(novoTotal, "R$ #,##0.00") & vbCrLf & vbCrLf & _
        "Confirma?", vbYesNo + vbQuestion, "Confirmar Alteração")
    
    If confirmacao = vbNo Then Exit Sub
    
    ' Atualizar os dados na lista
    Me.lstSelecionados.List(indiceItem, 4) = qtdDisplayNova
    Me.lstSelecionados.List(indiceItem, 5) = Format(descontoAtual, "R$ #,##0.00")
    Me.lstSelecionados.List(indiceItem, 6) = Format(novoTotal, "R$ #,##0.00")
    
    Call CalcularTodosOsTotais
    
    MsgBox "QUANTIDADE ALTERADA!" & vbCrLf & vbCrLf & _
           nomeItem & vbCrLf & _
           "Quantidade: " & qtdRealNova & " produto(s)" & vbCrLf & _
           "Display: " & qtdDisplayNova & " UN" & vbCrLf & _
           "Total: " & Format(novoTotal, "R$ #,##0.00"), vbInformation
    
    Exit Sub
TratarErro:
    MsgBox "Erro ao alterar quantidade: " & Err.Description, vbCritical
End Sub

' === BOTAO ADICIONAR MODIFICADO PARA PDV MADEIREIRA ===
Private Sub btnAdicionar_Click()
    On Error GoTo TratarErro
    
    If Me.lstProdutos.ListIndex < 0 Then
        MsgBox "Selecione um produto.", vbExclamation
        Exit Sub
    End If
    
    Dim i As Long, idProduto As String
    idProduto = CStr(Me.lstProdutos.List(Me.lstProdutos.ListIndex, 0))
    
    ' Verificar se produto já foi adicionado
    For i = 0 To Me.lstSelecionados.ListCount - 1
        If CStr(Me.lstSelecionados.List(i, 0)) = idProduto Then
            MsgBox "Produto já adicionado.", vbInformation
            Exit Sub
        End If
    Next i
    
    ' Buscar preço do produto na planilha
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Worksheets("Produtos")
    
    Dim precoUnitario As Currency
    Dim linhaAtual As Long
    
    For linhaAtual = 2 To ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
        If CStr(ws.Cells(linhaAtual, 1).Value) = idProduto Then
            precoUnitario = ws.Cells(linhaAtual, 6).Value
            Exit For
        End If
    Next linhaAtual
    
    ' Obter quantidade
    Dim qtdReal As Long
    qtdReal = CLng(Me.txtQuantidade.Value)
    
    If qtdReal < QTD_MINIMA Or qtdReal > QTD_MAXIMA Then
        MsgBox "Quantidade deve estar entre " & QTD_MINIMA & " e " & QTD_MAXIMA, vbExclamation
        Exit Sub
    End If
    
    Dim qtdDisplay As String
    Dim subtotalItem As Currency
    
    qtdDisplay = FormatarQuantidadePDV(qtdReal)
    subtotalItem = precoUnitario * qtdReal
    
    ' Adicionar item na lista
    With Me.lstSelecionados
        .AddItem
        .List(.ListCount - 1, 0) = idProduto
        .List(.ListCount - 1, 1) = Me.lstProdutos.List(Me.lstProdutos.ListIndex, 1)
        .List(.ListCount - 1, 2) = "UN"
        .List(.ListCount - 1, 3) = Format(precoUnitario, "R$ #,##0.00")
        .List(.ListCount - 1, 4) = qtdDisplay
        .List(.ListCount - 1, 5) = "R$ 0,00"
        .List(.ListCount - 1, 6) = Format(subtotalItem, "R$ #,##0.00")
    End With
    
    Call CalcularTodosOsTotais
    
    ' Resetar quantidade
    Me.txtQuantidade.Value = QTD_MINIMA
    
    MsgBox "Produto adicionado!" & vbCrLf & vbCrLf & _
           "Quantidade: " & qtdDisplay & " UN (" & qtdReal & " produtos)" & vbCrLf & _
           "Preço: " & Format(precoUnitario, "R$ #,##0.00") & " cada" & vbCrLf & _
           "Total: " & Format(subtotalItem, "R$ #,##0.00"), vbInformation
    
    Exit Sub
TratarErro:
    MsgBox "Erro ao adicionar produto: " & Err.Description, vbCritical
End Sub

    On Error GoTo TratarErro

    Dim qtdReal As Long

' === FUNCAO CENTRAL: CALCULA TODOS OS TOTAIS (CORRIGIDA) ===
Private Sub CalcularTodosOsTotais()
    On Error Resume Next
    
    Dim i As Long
    Dim precoUnit As Currency
    Dim qtdReal As Long
    Dim desconto As Currency
    Dim valorFinalItem As Currency
    Dim subtotalItem As Currency
    
    totalSubtotal = 0
    totalDescontos = 0
    totalFinal = 0
    quantidadeItens = Me.lstSelecionados.ListCount
    
    For i = 0 To quantidadeItens - 1
        precoUnit = TextoParaMoeda(CStr(Me.lstSelecionados.List(i, 3)))
        qtdReal = QuantidadePDVParaReal(CStr(Me.lstSelecionados.List(i, 4)))
        desconto = TextoParaMoeda(CStr(Me.lstSelecionados.List(i, 5)))
        
        ' CORREÇÃO PRINCIPAL: Calcular subtotal baseado na quantidade real
        subtotalItem = precoUnit * qtdReal
        valorFinalItem = subtotalItem - desconto
        
        totalSubtotal = totalSubtotal + subtotalItem
        totalDescontos = totalDescontos + desconto
        totalFinal = totalFinal + valorFinalItem
        
        ' DEBUG: Mostrar cálculos no console
        Debug.Print "Item " & i & ": Preço=" & precoUnit & " x Qtd=" & qtdReal & " = " & subtotalItem
    Next i
    
    Call AtualizarLabelsDoTotais
    On Error GoTo 0
End Sub

        Me.lblTotal.Caption = Format(totalFinal, "R$ #,##0.00")

' === ENVIO PARA FORMULARIO PRINCIPAL ===
Private Sub btnEnviarParaProdutos_Click()
    On Error GoTo ErroEnvio
    
    Dim formPDV As Object
    Dim frm As Object
    Dim formEncontrado As Boolean
    Dim listaDestino As Object
    Dim nomeControle As String
    Dim i As Long, n As Long
    
    If Me.lstSelecionados.ListCount = 0 Then
        MsgBox "Adicione produtos antes de enviar.", vbExclamation
        Exit Sub
    End If

    formEncontrado = False

    For Each frm In VBA.UserForms
        Debug.Print "Formulário encontrado: " & TypeName(frm)

            Set formPDV = frm
            formEncontrado = True
            Debug.Print "Formulário frmPDVPrincipal encontrado!"
            Exit For
        End If

               "Abra primeiro o formulário principal.", vbCritical

        If Not listaDestino Is Nothing Then
            nomeControle = "produtosv1"
            Debug.Print "Lista produtosv1 encontrada!"
        End If
    End If
    On Error GoTo ErroEnvio

    If listaDestino Is Nothing Then
        MsgBox "Lista de produtos não encontrada no formulário principal." & vbCrLf & _
               "Verifique se existe 'produtosv1' ou 'produtosv2'.", vbCritical
        Exit Sub
    End If

    End With

        With listaDestino
            .AddItem Me.lstSelecionados.List(i, 0)
            n = .ListCount - 1
            
            .List(n, 1) = Me.lstSelecionados.List(i, 1)
            .List(n, 2) = Me.lstSelecionados.List(i, 2)
            .List(n, 3) = Me.lstSelecionados.List(i, 3)
            .List(n, 4) = Me.lstSelecionados.List(i, 4)
            .List(n, 5) = Me.lstSelecionados.List(i, 5)
            .List(n, 6) = Me.lstSelecionados.List(i, 6)
        End With

    Debug.Print "Total final enviado: " & Format(totalFinal, "R$ #,##0.00")

        If .ListCount > 0 Then

' ===== GERAR NÚMERO DO PEDIDO (DETECÇÃO AUTOMÁTICA) =====
Private Function GerarNumeroPedido(ws As Worksheet) As String
    Dim ultimoNumero As Long
    Dim proximoNumero As Long
    
    ' Verificar números existentes em AMBOS os lados
    Dim numeroEsquerdo As Long
    Dim numeroDireito As Long
    
    ' Verificar lado esquerdo (célula A5)
    numeroEsquerdo = ExtrairNumeroPedido(ws.Range("A5").Value)
    
    ' Verificar lado direito (célula K5)
    numeroDireito = ExtrairNumeroPedido(ws.Range("K5").Value)
    
    ' Encontrar o maior número entre os dois lados
    ultimoNumero = Application.WorksheetFunction.Max(numeroEsquerdo, numeroDireito, 0)
    
    ' Se ambos estiverem vazios, começa do 1
    If ultimoNumero = 0 Then
        proximoNumero = 1
    Else
        proximoNumero = ultimoNumero + 1
    End If
    
    ' Atualizar contador na célula Z1 (para referência futura)
    ws.Range("Z1").Value = proximoNumero
    
    ' Formatar número do pedido com 6 dígitos
    GerarNumeroPedido = Format(proximoNumero, "000000")
End Function

' ===== EXTRAIR NÚMERO DO TEXTO DO PEDIDO =====
Private Function ExtrairNumeroPedido(texto As String) As Long
    If texto = "" Then
        ExtrairNumeroPedido = 0
        Exit Function
    End If
    
    ' Procurar sequência de 6 dígitos no texto
    Dim regex As Object
    Dim matches As Object
    Dim numero As String
    
    Set regex = CreateObject("VBScript.RegExp")
    With regex
        .Global = True
        .Pattern = "\d{6}" ' 6 dígitos
    End With
    
    Set matches = regex.Execute(texto)
    
    If matches.Count > 0 Then
        numero = matches(0)
        If IsNumeric(numero) Then
            ExtrairNumeroPedido = CLng(numero)
        Else
            ExtrairNumeroPedido = 0
        End If
    Else
        ExtrairNumeroPedido = 0
    End If
End Function

' ===== PREENCHIMENTO COMPLETO DO FORMULÁRIO ESQUERDO (CORRIGIDO) =====
Private Sub PreencherFormularioEsquerdoCompleto(ws As Worksheet, numeroPedido As String)
    ' Limpar área primeiro
    ws.Range("B7:H9, A11:J21, B22:J25").ClearContents
    
    ' === NÚMERO DO PEDIDO (ESQUERDO) ===
    ws.Range("A5").Value = "PEDIDO DE VENDA " & numeroPedido
    
    ' === DADOS DO CLIENTE (ESQUERDO) ===
    ws.Range("B7").Value = Me.txtNome.Value
    
    ' Endereço completo
    Dim enderecoCompleto As String
    enderecoCompleto = Me.txtEnder.Value
    If Trim(Me.txtnumero.Value) <> "" Then
        enderecoCompleto = enderecoCompleto & ", " & Me.txtnumero.Value
    End If
    ws.Range("B8").Value = enderecoCompleto
    
    ws.Range("B9").Value = Me.txtCPF.Value
    ws.Range("E9").Value = Me.cCidade.Value
    ws.Range("F8").Value = Me.cbairro1.Value
    ws.Range("G9").Value = "PE"
    ws.Range("H9").Value = Me.txtCEP.Value
    ws.Range("H5").Value = Format(Now, "dd/mm/yyyy hh:mm")
    ws.Range("J5").Value = Format(Now, "dd/mm/yyyy hh:mm")
    
    ' === PRODUTOS (ESQUERDO) ===
    Call PreencherProdutosEsquerdo(ws)
    
    ' === RODAPÉ (ESQUERDO) ===
    ws.Range("B22").Value = "VENDEDOR SISTEMA"
    ws.Range("B23").Value = "SITUAÇÃO ATUAL"
    ws.Range("C24").Value = Me.cPagamento.Value
    
    If Me.cData.Value <> "" Then
        ws.Range("F23").Value = Me.cData.Value
    Else
        ws.Range("F23").Value = Format(Date + 1, "dd/mm/yyyy")
    End If
    
    ' Valores (CORRIGIDO - não duplicar R$)
    Dim totalProdutos As Double
    totalProdutos = CalcularTotalProdutos()
    
    ws.Range("J22").Value = totalProdutos
    ws.Range("J23").Value = 0
    ws.Range("J24").Value = 0
    ws.Range("J25").Value = totalProdutos
    
    ' Formatar como moeda
    ws.Range("J22:J25").NumberFormat = "R$ #,##0.00"
    
    ' Formatar número do pedido
    With ws.Range("A5")
        .Font.Bold = True
        .Font.Size = 12
        .HorizontalAlignment = xlLeft
    End With
End Sub

Private Sub PreencherProdutosEsquerdo(ws As Worksheet)
    Dim i As Long, linha As Long
    linha = 11 ' Começa na linha 11
    
    For i = 0 To Me.produtosv2.ListCount - 1
        If linha > 21 Then Exit For ' Máximo 10 produtos
        
        ' Coluna A: Referência
        ws.Cells(linha, 1).Value = Me.produtosv2.List(i, 0)
        
        ' Coluna B: Descrição
        ws.Cells(linha, 2).Value = Me.produtosv2.List(i, 1)
        
        ' Coluna C: Unidade
        ws.Cells(linha, 3).Value = Me.produtosv2.List(i, 2)
        
        ' Coluna D: Valor Unitário (apenas número, formatação automática)
        ws.Cells(linha, 4).Value = TextoParaMoeda(Me.produtosv2.List(i, 3))
        ws.Cells(linha, 4).NumberFormat = "R$ #,##0.00"
        
        ' Coluna E: Quantidade (manter formato PDV)
        ws.Cells(linha, 5).Value = Me.produtosv2.List(i, 4)
        
        ' Coluna F: Desconto (apenas número, formatação automática)
        ws.Cells(linha, 6).Value = TextoParaMoeda(Me.produtosv2.List(i, 5))
        ws.Cells(linha, 6).NumberFormat = "R$ #,##0.00"
        
        ' Coluna G: Valor Total (apenas número, formatação automática)
        ws.Cells(linha, 7).Value = TextoParaMoeda(Me.produtosv2.List(i, 6))
        ws.Cells(linha, 7).NumberFormat = "R$ #,##0.00"
        
        linha = linha + 1
    Next i
End Sub

' ===== FUNÇÃO PARA CONVERTER TEXTO EM MOEDA (APRIMORADA) =====
Private Function TextoParaMoeda(texto As String) As Double
    On Error Resume Next
    
    If texto = "" Or IsNull(texto) Then
        TextoParaMoeda = 0
        Exit Function
    End If
    
    Dim valor As String
    valor = Trim(CStr(texto))
    
    ' Remove "R$" e espaços
    valor = Replace(valor, "R$", "")
    valor = Replace(valor, " ", "")
    valor = Trim(valor)
    
    ' Se está vazio após limpeza
    If valor = "" Then
        TextoParaMoeda = 0
        Exit Function
    End If
    
    ' Tratar formatação brasileira (1.234,56)
    If InStr(valor, ".") > 0 And InStr(valor, ",") > 0 Then
        ' Formato brasileiro: 1.234,56
        valor = Replace(valor, ".", "")  ' Remove separador de milhares
        valor = Replace(valor, ",", ".") ' Troca vírgula por ponto decimal
    ElseIf InStr(valor, ",") > 0 And Len(valor) - InStr(valor, ",") = 2 Then
        ' Apenas vírgula decimal: 123,45
        valor = Replace(valor, ",", ".")
    End If
    
    ' Converter para número
    If IsNumeric(valor) Then
        TextoParaMoeda = CDbl(valor)
    Else
        TextoParaMoeda = 0
    End If
    
    On Error GoTo 0
End Function

Private Sub btnLimparPlanilha_Click()
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Worksheets("marialuiza(1)")
    
    If MsgBox("Deseja limpar TODOS os dados da planilha?" & vbCrLf & _
             "Isso apagará todos os formulários!", vbYesNo + vbWarning, "Limpar Tudo") = vbYes Then
        ' Limpar formulário esquerdo
        ws.Range("B7:H9, A11:J21, B22:J25").ClearContents
        ' Limpar formulário direito
        ws.Range("M7:T9, L11:V21, M22:V25").ClearContents
        
        MsgBox "Planilha limpa com sucesso!", vbInformation
    End If
End Sub

Private Sub btnReiniciarContador_Click()

    Set ws = ThisWorkbook.Worksheets("marialuiza(1)")
    
    If MsgBox("Deseja reiniciar o contador de pedidos para 000001?" & vbCrLf & _

    End If
End Sub

' ===== FUNÇÃO PARA DEFINIR MANUALMENTE =====
Private Sub btnDefinirNumero_Click()
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Worksheets("marialuiza(1)")
    
    Dim novoNumero As String
    novoNumero = InputBox("Digite o próximo número do pedido (6 dígitos):", "Definir Número do Pedido", "000001")
    
    If novoNumero <> "" Then
        If IsNumeric(novoNumero) And Len(novoNumero) <= 6 Then
            Dim numero As Long
            numero = CLng(novoNumero)
            
            ws.Range("Z1").Value = numero
            
            MsgBox "Próximo número definido para: " & Format(numero, "000000"), vbInformation
        Else
            MsgBox "Por favor, digite um número válido de até 6 dígitos!", vbExclamation
        End If
    End If
End Sub
