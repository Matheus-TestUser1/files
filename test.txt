' ===== INICIALIZAÇÃO DO SISTEMA =====
Private Sub InicializarSistemaNumeracao()

TratarErro:

End Sub

' ===== OBTER OU CRIAR PLANILHA CONTROLE =====
Private Function ObterOuCriarPlanilhaControle() As Worksheet
    On Error GoTo TratarErro
    
    Dim ws As Worksheet
    
    ' Tentar localizar planilha existente
    On Error Resume Next
    Set ws = ThisWorkbook.Sheets(PLANILHA_CONTROLE)
    On Error GoTo TratarErro
    
    If Not ws Is Nothing Then
        Debug.Print "? Planilha controle encontrada: " & PLANILHA_CONTROLE
        Set ObterOuCriarPlanilhaControle = ws
        Exit Function
    End If
    
    ' Criar nova planilha
    Debug.Print "? Planilha não encontrada. Criando..."
    
    Set ws = ThisWorkbook.Worksheets.Add(Before:=ThisWorkbook.Sheets(1))
    ws.Name = PLANILHA_CONTROLE
    Call ConfigurarEstruturaPlanilha(ws)
    
    Set ObterOuCriarPlanilhaControle = ws
    Debug.Print "? Planilha controle criada com sucesso"
    
    Exit Function
    
TratarErro:
    Debug.Print "? ERRO ObterOuCriarPlanilhaControle: " & Err.Description
    Set ObterOuCriarPlanilhaControle = Nothing
End Function

' ===== VERIFICAR E CORRIGIR NUMERAÇÃO =====
Private Sub VerificarECorrigirNumeracao(ws As Worksheet)
    On Error GoTo TratarErro
    
    Dim precisaSalvar As Boolean
    precisaSalvar = False
    
    Debug.Print ""
    Debug.Print "?? Verificando integridade da numeração..."
    Debug.Print "-----------------------------------------"
    
    Dim valorA1 As Variant
    valorA1 = ws.Range("A1").Value
    
    Debug.Print "Valor atual em A1: [" & valorA1 & "]"
    Debug.Print "Tipo: " & TypeName(valorA1)
    Debug.Print "IsEmpty: " & IsEmpty(valorA1)
    Debug.Print "IsNull: " & IsNull(valorA1)
    Debug.Print "IsNumeric: " & IsNumeric(valorA1)
    
    ' ===== DETECÇÃO E CORREÇÃO DE PROBLEMAS =====
    If IsEmpty(valorA1) Or IsNull(valorA1) Or valorA1 = "" Or Not IsNumeric(valorA1) Then
        Debug.Print "? PROBLEMA: Célula A1 vazia ou inválida"
        Call RecuperarOuIniciarNumeracao(ws)
        precisaSalvar = True
        
    ElseIf CLng(valorA1) <= 0 Then
        Debug.Print "? PROBLEMA: Número zero ou negativo detectado [" & valorA1 & "]"
        Call RecuperarOuIniciarNumeracao(ws)
        precisaSalvar = True
        
    Else
        Debug.Print "? Número válido: " & Format(CLng(valorA1), "000000")
    End If
    
    ' ===== ATUALIZAR INFORMAÇÕES =====
    ws.Range("B8").Value = Format(ws.Range("A1").Value, "000000")
    ws.Range("B9").Value = ObterTotalImpressoes(ws)
    ws.Range("B10").Value = Format(Now, "dd/mm/yyyy hh:mm:ss")
    
    If precisaSalvar Then
        ThisWorkbook.save
        Debug.Print "? Numeração corrigida e salva"
    End If
    
    Debug.Print "-----------------------------------------"
    Debug.Print "? Verificação concluída"
    Debug.Print "? Próximo pedido será: " & Format(ws.Range("A1").Value, "000000")
    Debug.Print ""
    
    Exit Sub
    
TratarErro:
    Debug.Print "? ERRO VerificarECorrigirNumeracao: " & Err.Description
End Sub

' ===== CORREÇÃO DA FUNÇÃO QUE DETECTA O ÚLTIMO NÚMERO =====
Private Function ObterUltimoNumeroHistorico(ws As Worksheet) As Long
    On Error GoTo TratarErro
    
    Dim ultimaLinha As Long
    ultimaLinha = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    
    Debug.Print "?? Verificando histórico - Última linha: " & ultimaLinha
    
    If ultimaLinha < 15 Then  ' Linha 15 é a primeira do histórico
        Debug.Print "?? Histórico vazio ou apenas cabeçalho"
        ObterUltimoNumeroHistorico = 0
        Exit Function
    End If
    
    Dim maxNumero As Long
    maxNumero = 0
    Dim i As Long, numTemp As Variant
    Dim encontrouNumeros As Boolean
    encontrouNumeros = False
    
    ' Verificar da última linha para cima (mais eficiente)
    For i = ultimaLinha To 15 Step -1
        numTemp = ws.Cells(i, 1).Value
        
        ' Debug detalhado
        Debug.Print "  Linha " & i & ": [" & numTemp & "] | Tipo: " & TypeName(numTemp)
        
        If Not IsEmpty(numTemp) And numTemp <> "" Then
            ' Tentar converter para número
            If IsNumeric(numTemp) Then
                Dim num As Long
                On Error Resume Next
                num = CLng(numTemp)
                On Error GoTo 0
                
                If num > 0 Then
                    If num > maxNumero Then
                        maxNumero = num
                        encontrouNumeros = True
                        Debug.Print "  ?? NOVO MÁXIMO ENCONTRADO: " & Format(num, "000000")
                    End If
                End If
            Else
                ' Tentar extrair número de string (ex: "000001")
                Dim texto As String
                texto = CStr(numTemp)
                texto = Replace(texto, " ", "")
                
                If IsNumeric(texto) Then
                    Dim numTexto As Long
                    On Error Resume Next
                    numTexto = CLng(texto)
                    On Error GoTo 0
                    
                    If numTexto > 0 And numTexto > maxNumero Then
                        maxNumero = numTexto
                        encontrouNumeros = True
                        Debug.Print "  ?? NÚMERO EXTRAÍDO DA STRING: " & Format(numTexto, "000000")
                    End If
                End If
            End If
        End If
        
        ' Otimização: se encontrou um número alto, pode parar
        If maxNumero > 1000 Then Exit For
    Next i
    
    If encontrouNumeros Then
        Debug.Print "?? ÚLTIMO NÚMERO DO HISTÓRICO: " & Format(maxNumero, "000000")
        ObterUltimoNumeroHistorico = maxNumero
    Else
        Debug.Print "?? NENHUM NÚMERO VÁLIDO ENCONTRADO NO HISTÓRICO"
        ObterUltimoNumeroHistorico = 0
    End If
    
    Exit Function
    
TratarErro:
    Debug.Print "?? ERRO ObterUltimoNumeroHistorico: " & Err.Description
    ObterUltimoNumeroHistorico = 0
End Function

' ===== CORREÇÃO DA FUNÇÃO RECUPERAR NUMERAÇÃO =====
Private Sub RecuperarOuIniciarNumeracao(ws As Worksheet)
    On Error Resume Next
    
    Debug.Print ""
    Debug.Print "?? INICIANDO RECUPERAÇÃO DE NUMERAÇÃO..."
    
    ' PRIMEIRO: Verificar se A1 já tem um número válido
    Dim valorAtual As Variant
    valorAtual = ws.Range("A1").Value
    
    Debug.Print "?? Valor atual em A1: [" & valorAtual & "]"
    
    ' Se A1 já tem um número válido, usar ele
    If Not IsEmpty(valorAtual) And IsNumeric(valorAtual) Then
        Dim numAtual As Long
        numAtual = CLng(valorAtual)
        
        If numAtual > 0 Then
            Debug.Print "?? USANDO NÚMERO EXISTENTE EM A1: " & Format(numAtual, "000000")
            Exit Sub
        End If
    End If
    
    ' SEGUNDO: Tentar recuperar do histórico
    Dim ultimoNumero As Long
    ultimoNumero = ObterUltimoNumeroHistorico(ws)
    
    If ultimoNumero > 0 Then
        ws.Range("A1").Value = ultimoNumero + 1
        Debug.Print "?? RECUPERADO DO HISTÓRICO: " & Format(ultimoNumero, "000000")
        Debug.Print "?? PRÓXIMO NÚMERO SERÁ: " & Format(ultimoNumero + 1, "000000")
    Else
        ' TERCEIRO: Iniciar do zero
        ws.Range("A1").Value = 1
        Debug.Print "?? HISTÓRICO VAZIO. INICIANDO EM: 000001"
    End If
    
    ' GARANTIR formatação correta
    ws.Range("A1").NumberFormat = "0"
    ws.Range("A1").Font.Bold = True
    ws.Range("A1").HorizontalAlignment = xlCenter
    
    ' ATUALIZAR DISPLAY
    ws.Range("B8").Value = Format(ws.Range("A1").Value, "000000")
    
    Debug.Print "?? NUMERAÇÃO CONFIGURADA: " & Format(ws.Range("A1").Value, "000000")
    
    On Error GoTo 0
End Sub

' ===== CORREÇÃO DA FUNÇÃO GERAR NÚMERO =====
Private Function GerarProximoNumeroPedido() As String
    On Error GoTo TratarErro
    
    Debug.Print ""
    Debug.Print "========================================="
    Debug.Print "?? GERANDO NOVO NÚMERO DE PEDIDO - v8.1"
    Debug.Print "========================================="
    
    Dim ws As Worksheet
    Set ws = ObterOuCriarPlanilhaControle()
    
    If ws Is Nothing Then
        Debug.Print "?? ERRO: Planilha controle não disponível"
        GerarProximoNumeroPedido = Format(proximoPedido, "000000")
        proximoPedido = proximoPedido + 1
        Exit Function
    End If
    
    ' VERIFICAR ANTES DE LER
    Call VerificarECorrigirNumeracao(ws)
    
    ' LER NÚMERO ATUAL COM VALIDAÇÃO
    Dim numeroAtual As Long
    numeroAtual = CLng(ws.Range("A1").Value)
    
    Debug.Print "?? NÚMERO ATUAL EM A1: " & numeroAtual
    Debug.Print "?? NÚMERO FORMATADO: " & Format(numeroAtual, "000000")
    
    ' VALIDAÇÃO CRÍTICA
    If numeroAtual <= 0 Then
        Debug.Print "?? PROBLEMA CRÍTICO: Número inválido! Forçando correção..."
        ws.Range("A1").Value = 1
        numeroAtual = 1
        Debug.Print "?? NÚMERO CORRIGIDO PARA: " & numeroAtual
    End If
    
    Dim numeroFormatado As String
    numeroFormatado = Format(numeroAtual, "000000")
    
    Debug.Print "?? NÚMERO GERADO: " & numeroFormatado
    
    ' ===== INCREMENTAR PARA PRÓXIMO =====
    Dim proximoNumero As Long
    proximoNumero = numeroAtual + 1
    
    Debug.Print "?? ATUALIZANDO A1 PARA: " & proximoNumero
    
    ' ATUALIZAR CÉLULA PRINCIPAL
    ws.Range("A1").Value = proximoNumero
    ws.Range("A1").NumberFormat = "0"
    
    ' ATUALIZAR DISPLAY
    ws.Range("B8").Value = Format(proximoNumero, "000000")
    ws.Range("B10").Value = Format(Now, "dd/mm/yyyy hh:mm:ss")
    
    ' ATUALIZAR VARIÁVEL GLOBAL
    proximoPedido = proximoNumero
    
    ' SALVAR IMEDIATAMENTE
    ThisWorkbook.save
    
    Debug.Print "?? PLANILHA ATUALIZADA E SALVA"
    Debug.Print "?? PRÓXIMO PEDIDO SERÁ: " & Format(proximoNumero, "000000")
    
    ' REGISTRAR NO HISTÓRICO
    Call RegistrarPedidoHistorico(ws, numeroFormatado)
    
    Debug.Print "========================================="
    Debug.Print "?? NÚMERO GERADO COM SUCESSO!"
    Debug.Print "   ATUAL: " & numeroFormatado
    Debug.Print "   PRÓXIMO: " & Format(proximoNumero, "000000")
    Debug.Print "========================================="
    Debug.Print ""
    
    GerarProximoNumeroPedido = numeroFormatado
    
    Exit Function
    
TratarErro:
    Debug.Print "?? ERRO CRÍTICO GerarProximoNumeroPedido: " & Err.Description
    
    ' FALLBACK SUPER SEGURO
    Static contadorFallback As Long
    If contadorFallback = 0 Then
        ' Tentar recuperar da planilha primeiro
        On Error Resume Next
        contadorFallback = ws.Range("A1").Value
        If contadorFallback <= 0 Then contadorFallback = 1
        On Error GoTo 0
    End If
    
    contadorFallback = contadorFallback + 1
    GerarProximoNumeroPedido = Format(contadorFallback, "000000")
    
    Debug.Print "?? USANDO FALLBACK: " & GerarProximoNumeroPedido
End Function

' ===== REGISTRAR NO HISTÓRICO =====
Private Sub RegistrarPedidoHistorico(ws As Worksheet, numeroPedido As String)
    On Error Resume Next
    
    Debug.Print "?? Registrando no histórico..."
    
    Dim ultimaLinha As Long
    ultimaLinha = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row + 1
    If ultimaLinha < 15 Then ultimaLinha = 15
    
    Debug.Print "   Linha do histórico: " & ultimaLinha
    
    With ws
        .Cells(ultimaLinha, 1).Value = numeroPedido
        .Cells(ultimaLinha, 2).Value = Format(Now, "dd/mm/yyyy hh:mm:ss")
        .Cells(ultimaLinha, 3).Value = ObterNomeCliente()
        .Cells(ultimaLinha, 4).Value = ObterTotalVenda()
        .Cells(ultimaLinha, 5).Value = "IMPRESSO"
        
        ' Formatação
        With .Range(.Cells(ultimaLinha, 1), .Cells(ultimaLinha, 5))
            .Borders.LineStyle = xlContinuous
            .HorizontalAlignment = xlCenter
        End With
        
        .Cells(ultimaLinha, 1).HorizontalAlignment = xlCenter
        .Cells(ultimaLinha, 1).Font.Bold = True
        
        ' Atualizar contador
        .Range("B9").Value = ultimaLinha - 14
    End With
    
    ThisWorkbook.save
    Debug.Print "? Histórico atualizado (Total: " & (ultimaLinha - 14) & " pedidos)"
    
    On Error GoTo 0
End Sub

Private Sub PreencherLadoFormulario(ws As Worksheet, numeroPedido As String, lado As String)

    Dim rngCPF As Range, rngCidade As Range, rngBairro As Range
    Dim rngUF As Range, rngCEP As Range, rngProdutos As Range
    Dim rngVendedor As Range, rngSituacao As Range, rngPagamento As Range
    Dim rngEntrega As Range, rngValorProdutos As Range, rngFrete As Range
    Dim rngDesconto As Range, rngTotal As Range
    Dim rngHora As Range, rngData As Range
    
    If UCase(lado) = "ESQUERDO" Then
        colOffset = 0
        Set rngCabecalho = ws.Range("C5")
        Set rngCliente = ws.Range("B7")
        Set rngEndereco = ws.Range("B8")
        Set rngCPF = ws.Range("B9")
        Set rngCidade = ws.Range("E9")
        Set rngBairro = ws.Range("F8")
        Set rngUF = ws.Range("G9")
        Set rngCEP = ws.Range("I9")
        Set rngProdutos = ws.Range("A11:J43")
        Set rngVendedor = ws.Range("B44")
        Set rngSituacao = ws.Range("B45")
        Set rngPagamento = ws.Range("C46")
        Set rngEntrega = ws.Range("F45")
        Set rngValorProdutos = ws.Range("J44")
        Set rngFrete = ws.Range("J45")
        Set rngDesconto = ws.Range("J46")
        Set rngTotal = ws.Range("J47")

        Set rngData = ws.Range("I5")

        Set rngEndereco = ws.Range("M8")
        Set rngCPF = ws.Range("M9")
        Set rngCidade = ws.Range("P9")

        Set rngDesconto = ws.Range("U46")

    Call PreencherTotaisCorrigido(rngValorProdutos, rngFrete, rngDesconto, rngTotal)
    
    On Error GoTo 0
End Sub

Private Sub PreencherProdutosLadoCorrigido(ws As Worksheet, rngProdutos As Range, colOffset As Long)
    On Error Resume Next
    
    Dim i As Long, linha As Long
    linha = rngProdutos.Row
    
    For i = 0 To Me.produtosv2.ListCount - 1
        If linha > rngProdutos.Row + rngProdutos.Rows.Count - 1 Then Exit For
        
        ' Limpar formatação
        ws.Cells(linha, 1 + colOffset).ClearFormats
        ws.Cells(linha, 2 + colOffset).ClearFormats
        ws.Cells(linha, 3 + colOffset).ClearFormats
        ws.Cells(linha, 4 + colOffset).ClearFormats
        ws.Cells(linha, 5 + colOffset).ClearFormats
        ws.Cells(linha, 6 + colOffset).ClearFormats
        ws.Cells(linha, 7 + colOffset).ClearFormats
        
        ' Preencher dados
        ws.Cells(linha, 1 + colOffset).Value = Me.produtosv2.List(i, 0)
        ws.Cells(linha, 2 + colOffset).Value = Me.produtosv2.List(i, 1)
        ws.Cells(linha, 3 + colOffset).Value = "UN"
        
        ' Valores numéricos
        ws.Cells(linha, 4 + colOffset).Value = CDbl(LimparValor(Me.produtosv2.List(i, 3)))
        ws.Cells(linha, 6 + colOffset).Value = CDbl(LimparValor(Me.produtosv2.List(i, 5)))
        ws.Cells(linha, 7 + colOffset).Value = CDbl(LimparValor(Me.produtosv2.List(i, 6)))
        
        ' Formatação
        ws.Cells(linha, 4 + colOffset).NumberFormat = """R$ ""#,##0.00"
        ws.Cells(linha, 6 + colOffset).NumberFormat = """R$ ""#,##0.00"
        ws.Cells(linha, 7 + colOffset).NumberFormat = """R$ ""#,##0.00"
        
        ' Quantidade
        Dim quantidadeTexto As String
        quantidadeTexto = Me.produtosv2.List(i, 4)
        
        If IsNumeric(quantidadeTexto) Then
            ws.Cells(linha, 5 + colOffset).Value = CDbl(quantidadeTexto)
            ws.Cells(linha, 5 + colOffset).NumberFormat = "#,##0.000"
        Else
            ws.Cells(linha, 5 + colOffset).Value = quantidadeTexto
        End If
        
        linha = linha + 1
    Next i
    
    On Error GoTo 0
End Sub

Private Sub PreencherTotaisCorrigido(rngValorProdutos As Range, rngFrete As Range, rngDesconto As Range, rngTotal As Range)
    Dim totalFinal As Currency
    totalFinal = ObterTotalFinalProdutos()
    
    ' Limpar
    rngValorProdutos.ClearFormats
    rngValorProdutos.ClearContents
    rngFrete.ClearFormats
    rngFrete.ClearContents
    rngDesconto.ClearFormats
    rngDesconto.ClearContents
    rngTotal.ClearFormats
    rngTotal.ClearContents
    
    ' Preencher
    rngValorProdutos.Value = CDbl(totalFinal)
    rngFrete.Value = 0
    rngDesconto.Value = 0
    rngTotal.Value = CDbl(totalFinal)
    
    ' Formatar
    rngValorProdutos.NumberFormat = """R$ ""#,##0.00"
    rngFrete.NumberFormat = """R$ ""#,##0.00"
    rngDesconto.NumberFormat = """R$ ""#,##0.00"
    rngTotal.NumberFormat = """R$ ""#,##0.00"
    
    Debug.Print "Total preenchido: " & totalFinal & " | Célula: " & rngTotal.Address
End Sub

' ===== SISTEMA DE CÁLCULO DE TOTAIS =====
Private Function LimparValor(valor As Variant) As Currency
    On Error GoTo TratarErro
    
    If IsEmpty(valor) Or IsNull(valor) Or valor = "" Then
        LimparValor = 0
        Exit Function
    End If
    
    If IsNumeric(valor) Then
        LimparValor = CCur(valor)
        Exit Function
    End If
    
    Dim texto As String
    texto = CStr(valor)
    texto = Replace(texto, "R$", "")
    texto = Replace(texto, "RR$", "")
    texto = Replace(texto, " ", "")
    texto = Replace(texto, ".", "")
    texto = Replace(texto, ",", ".")
    texto = Trim(texto)
    
    If texto = "" Then
        LimparValor = 0
        Exit Function
    End If
    
    If IsNumeric(texto) Then
        LimparValor = CCur(texto)
    Else
        LimparValor = 0
    End If
    
    Exit Function
    
TratarErro:
    LimparValor = 0
End Function

Private Function CalcularTotalProdutosColuna6() As Currency
    Dim totalCalculadoColuna6 As Currency
    totalCalculadoColuna6 = 0
    
    Dim indiceItem As Long
    For indiceItem = 0 To Me.produtosv2.ListCount - 1
        totalCalculadoColuna6 = totalCalculadoColuna6 + LimparValor(Me.produtosv2.List(indiceItem, 6))
    Next indiceItem
    
    CalcularTotalProdutosColuna6 = totalCalculadoColuna6
End Function

Private Function ObterTotalFinalProdutos() As Currency
    ObterTotalFinalProdutos = CalcularTotalProdutosColuna6()
End Function

' ===== FORMATAÇÃO DE CEP =====
Private Sub txtCEP_Change()
    Dim cep As String, CEP2 As String, CEP3 As String
    Dim i As Integer, j As Integer, N As Integer

    cep = txtCEP.Value
    txtCEP.MaxLength = 9
    i = Len(cep)

    For j = 1 To i
        If IsNumeric(Mid(cep, j, 1)) Then
            CEP2 = CEP2 & Mid(cep, j, 1)
        End If
    Next
    
    i = Len(CEP2)
    For j = 1 To i
        CEP3 = CEP3 & Mid(CEP2, j, 1)
        If j = 5 Then
            N = Len(CEP3) - 1
            CEP3 = Left(CEP3, N) & "-" & Right(CEP3, 1)
        End If
    Next

    txtCEP.Value = CEP3
    txtCEP.SelStart = Len(txtCEP.Value)
End Sub

    If (KeyAscii < 48 Or KeyAscii > 57) And KeyAscii <> 8 And KeyAscii <> 46 Then
        KeyAscii = 0
    End If
End Sub

' ===== VALIDAÇÃO DE CPF =====
Private Sub txtCpf_Change()
    Dim cpf As String, CPF2 As String, CPF3 As String
    Dim i As Integer, j As Integer, N As Integer

    cpf = txtCPF.Value
    txtCPF.MaxLength = 14
    i = Len(cpf)

    For j = 1 To i
        If IsNumeric(Mid(cpf, j, 1)) Then
            CPF2 = CPF2 & Mid(cpf, j, 1)
        End If
    Next
    
    i = Len(CPF2)
    For j = 1 To i
        CPF3 = CPF3 & Mid(CPF2, j, 1)
        If j = 4 Or j = 7 Then
            N = Len(CPF3) - 1
            CPF3 = Left(CPF3, N) & "." & Right(CPF3, 1)
        ElseIf j = 10 Then
            N = Len(CPF3) - 1
            CPF3 = Left(CPF3, N) & "-" & Right(CPF3, 1)
        End If
    Next

    txtCPF.Value = CPF3
End Sub
